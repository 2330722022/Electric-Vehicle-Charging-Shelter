    public void controlDevice(String property, boolean value) {
        new Thread(new Runnable() {
            @Override
            public void run() {
                try {
                    URL url = new URL("https://iot-api.heclouds.com/thingmodel/set-device-property");
                    HttpURLConnection connection = (HttpURLConnection) url.openConnection();
                    
                    connection.setRequestMethod("POST");
                    connection.setConnectTimeout(5000);
                    connection.setReadTimeout(5000);
                    connection.setRequestProperty("Content-Type", "application/json");
                    connection.setRequestProperty("authorization", ok_key);
                    connection.setDoOutput(true);
                    
                    JSONObject jsonBody = new JSONObject();
                    jsonBody.put("product_id", Pe_ID);
                    jsonBody.put("device_name", Device_Name);
                    
                    JSONObject params = new JSONObject();
                    params.put(property, value);
                    jsonBody.put("params", params);
                    
                    String requestBody = jsonBody.toString();
                    
                    OutputStream os = connection.getOutputStream();
                    os.write(requestBody.getBytes("UTF-8"));
                    os.flush();
                    os.close();
                    
                    int responseCode = connection.getResponseCode();
                    
                    if (responseCode == 200) {
                        InputStream is = connection.getInputStream();
                        BufferedReader br = new BufferedReader(new InputStreamReader(is));
                        StringBuilder response = new StringBuilder();
                        String line;
                        while ((line = br.readLine()) != null) {
                            response.append(line);
                        }
                        
                        String jsonResponse = response.toString();
                        JSONObject result = new JSONObject(jsonResponse);
                        int code = result.optInt("code", -1);
                        
                        runOnUiThread(new Runnable() {
                            @Override
                            public void run() {
                                if (code == 0) {
                                    if (property.equals("led")) {
                                        led_state = value;
                                    } else if (property.equals("beep")) {
                                        beep_state = value;
                                    }
                                    
                                    updateButtonStates();
                                    
                                    String deviceName = property.equals("led") ? "LED" : "蜂鸣器";
                                    
                                    Toast.makeText(MainActivity.this, 
                                        deviceName + (value ? " 开启成功" : " 关闭成功"), 
                                        Toast.LENGTH_SHORT).show();
                                } else {
                                    String msg = result.optString("msg", "未知错误");
                                    Toast.makeText(MainActivity.this, 
                                        "控制失败: " + msg, 
                                        Toast.LENGTH_LONG).show();
                                }
                            }
                        });
                        
                        connection.disconnect();
                    } else {
                        runOnUiThread(new Runnable() {
                            @Override
                            public void run() {
                                Toast.makeText(MainActivity.this, 
                                    "请求失败: HTTP " + responseCode, 
                                    Toast.LENGTH_LONG).show();
                            }
                        });
                    }
                } catch (Exception e) {
                    e.printStackTrace();
                    
                    runOnUiThread(new Runnable() {
                        @Override
                        public void run() {
                            Toast.makeText(MainActivity.this, 
                                "控制异常: " + e.getMessage(), 
                                Toast.LENGTH_LONG).show();
                        }
                    });
                }
            }
        }).start();
    }

    void DealJsonData(String JSON) throws JSONException {
        JSONObject jsonObject = new JSONObject(JSON);
        
        int code = jsonObject.optInt("code", -1);
        if (code != 0) {
            isDeviceOnline = false;
            return;
        }
        
        JSONArray data = jsonObject.optJSONArray("data");
        
        if (data == null) {
            isDeviceOnline = false;
            return;
        }
        
        for (int i = 0; i < data.length(); i++) {
            JSONObject value = data.getJSONObject(i);
            String identifier = value.optString("identifier");
            String val = value.optString("value");
            
            if(identifier.equals("temperature")) {
                temperature_value = val;
            } else if(identifier.equals("humidity")) {
                humidity_value = val;
            } else if(identifier.equals("light")) {
                light_value = val;
            } else if(identifier.equals("alarm_state")) {
                alarm_state = val.equals("1") || val.equalsIgnoreCase("true");
            } else if(identifier.equals("led")) {
                led_state = val.equals("1") || val.equalsIgnoreCase("true");
            } else if(identifier.equals("beep")) {
                beep_state = val.equals("1") || val.equalsIgnoreCase("true");
            }
        }
        
        isDeviceOnline = true;
    }
